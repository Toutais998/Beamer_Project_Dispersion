\englishtitle{Mechanical Constraints Behind Microscope Objectives}
\begin{frame}{物镜设计的光机协同约束}\label{2.1}
    \framerefbox{Zhang, Y. et al. Advanced Optical Technologies (2019). 10.1515/aot-2019-0002}
    \small
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth,height=0.52\textheight,keepaspectratio]{Figure/Zhang/ZhangFig_01.jpg}
                \caption{显微物镜设计中的机械边界与远心光瞳}
                \label{fig:objective_mechanics_main}
            \end{figure}
        \end{column}
        \hfill
        \begin{column}{.48\textwidth}
            \textbf{三个关键机械约束：}
            \begin{itemize}
                \item \alert{齐焦距}：样品面到物镜肩部的距离，限制轴向可用空间。
                \item \alert{螺纹直径}：限制后端光束、出瞳尺寸和镜组外径。
                \item \alert{远心光瞳}：保证倍率与照明稳定，便于和显微镜筒镜匹配。
            \end{itemize}

            \begin{block}{设计含义}
                高 NA、大视场、长工作距和平场并不会自然兼容；它们会同时挤压长度、口径、材料和装调预算。
            \end{block}

            \alert{因此，物镜设计应先将应用目标转化为机械边界与一阶规格，再开展具体光学结构设计。}
        \end{column}
    \end{columns}
\end{frame}

\subsection{公差与可制造闭环}
\label{sec:2.3}

\englishtitle{Tolerance Analysis}
\begin{frame}{为什么名义设计之后必须做公差分析}
\label{2.8}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{block}{名义设计的限制}
                名义设计默认曲率、厚度、折射率、偏心、倾斜都精确无误；真实制造中这些参数都会偏离。没有公差分析，就无法判断样品做出来以后还能不能满足指标。
            \end{block}

            \textbf{公差分析回答四个问题：}
            \begin{enumerate}
                \item 哪些面或参数最敏感？
                \item 当前供应商能力是否足够？
                \item 是否需要设置补偿器？
                \item 预计良率与成本是否可接受？
            \end{enumerate}
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \scriptsize
            \begin{table}
                \centering
                \renewcommand{\arraystretch}{1.28}
                \resizebox{\linewidth}{!}{%
