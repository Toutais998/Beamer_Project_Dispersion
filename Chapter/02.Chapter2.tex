\section{Zemax 仿真主线：从诊断到可制造}

\subsection{把理论变量映射到 Zemax 工具}

\englishtitle{OpticStudio as a Closed Loop}
\begin{frame}{Zemax 中的完整闭环}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{block}{工作流不是“打开很多分析窗口”}
                更稳妥的流程是：先用一阶规格把系统放对，再用 Ray Fan 和 OPD 找主要像差，用 Merit Function 改变量，最后用 MTF/PSF 和公差分析确认能否交付。
            \end{block}

            \begin{center}
                \fbox{\textbf{规格}}\quad
                $\rightarrow$\quad
                \fbox{\textbf{建模}}\quad
                $\rightarrow$\quad
                \fbox{\textbf{诊断}}\quad
                $\rightarrow$\quad
                \fbox{\textbf{优化}}\quad
                $\rightarrow$\quad
                \fbox{\textbf{制造}}
            \end{center}

            \bigbreak
            \textbf{核心判断：}每一次优化都要能回答“我正在压低哪一种物理误差，以及它会如何改善像质”。
        \end{column}

\englishtitle{Design Specification Before Optimization}
\begin{frame}{优化前先写清楚规格}
    \begin{columns}[T]
        \begin{column}{.45\textwidth}
            \textbf{一阶规格决定设计空间：}
            \begin{itemize}
                \item 焦距、放大率、工作距离、后焦距
                \item 孔径、$F/\#$、NA、入瞳位置
                \item 视场角、像面尺寸、探测器像元
                \item 波长范围、权重、是否单色照明
                \item 总长、镜片数量、外径、封装空间
                \item 成本、材料目录、加工等级、装调自由度
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{.55\textwidth}
            \begin{block}{System Explorer 中先确定}
                \begin{itemize}
                    \item \textbf{Aperture}：Entrance Pupil、Image Space NA 或 $F/\#$
                    \item \textbf{Fields}：角度、物高或像高
                    \item \textbf{Wavelengths}：主波长、权重、色差控制范围
                    \item \textbf{Units}：长度单位和玻璃目录
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{itemize}
                \item 先低权重锁定一阶目标，确认系统没有跑偏。
                \item 再提高关键视场、关键波长和关键频率权重。

        \end{column}

            \begin{itemize}
                \item 不重复约束同一物理量，避免优化器被冲突目标困住。
                \item 制造约束要从早期保留，不要最后才补。
            \end{itemize}
        \end{column}

\subsection{Zernike、PSF 与 MTF：确认优化是否真的改善图像}

\englishtitle{Wavefront-Based Optimization}
\begin{frame}{用 Zernike 和 RMS Wavefront 连接诊断与优化}
    \begin{columns}[T]
        \begin{column}{.52\textwidth}
            \textbf{推荐使用场景：}
            \begin{itemize}
                \item 系统接近衍射极限，需要从几何像差转向波前质量。
                \item 希望分离 defocus、astigmatism、coma、spherical 等模式。
                \item 后续需要和干涉仪测量、装调补偿或自由曲面拟合对接。
            \end{itemize}

            \begin{equation}
                S.R. \approx \exp\!\left[-\left(\frac{2\pi}{\lambda}OPD_{\text{rms}}\right)^2\right]
                \label{eq:strehl_zemax}
            \end{equation}

            \alert{当 RMS OPD 降低时，Strehl Ratio 通常会提高，PSF 主峰能量更集中。}
        \end{column}
        \hfill
        \begin{column}{.48\textwidth}
            \begin{block}{Zemax 中的做法}
                \begin{enumerate}
                    \item 查看 Wavefront Map 和 OPD Fan。
                    \item 输出 Zernike Standard Coefficients。
                    \item 去除 piston/tilt/defocus 后判断剩余主导项。
                    \item 在 Merit Function 中加入 RMS Wavefront 或 Zernike 相关约束。
                \end{enumerate}
            \end{block}

            \begin{block}{注意}
                不要把 Zernike 系数单独当作最终像质；它负责解释波前，最终仍需用 PSF、MTF 或圈入能量确认成像后果。
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Image-Quality Metrics}
\begin{frame}{像质评价：Ray Fan 之后看什么}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{block}{几何光学指标}
                \begin{itemize}
                    \item \textbf{Ray Fan}：最快定位像差类型。
                    \item \textbf{Spot Diagram}：看几何弥散斑是否落入像元。
                    \item \textbf{Distortion}：看主光线成像比例是否失真。
                \end{itemize}
            \end{block}

            \begin{block}{波前指标}
                \begin{itemize}
                    \item \textbf{OPD PV/RMS}：直接关联 Strehl Ratio。
                    \item \textbf{Zernike}：把 OPD 拆成可解释模式。
                \end{itemize}
            \end{block}
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \begin{block}{图像指标}
                \begin{itemize}
                    \item \textbf{PSF}：点像扩散成什么形状。
                    \item \textbf{MTF}：某空间频率还能传递多少对比度。
                    \item \textbf{Encircled Energy}：目标像元内能收进多少能量。
                \end{itemize}
            \end{block}

            \begin{alertblock}{传感器系统的检查}
                不只看 diffraction limit；还要把像元尺寸、Nyquist 频率、目标线对频率和照度要求一起放进评价标准。
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\subsection{材料、色差与玻璃替换}

\englishtitle{Glass Choice as Chromatic Aberration Control}
\begin{frame}{玻璃选择：色差优化的核心变量}
    \begin{columns}[T]
        \begin{column}{.44\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=.95\textwidth,height=.48\textheight,keepaspectratio]{Figure/Edmund/aberrations-5v2.pdf}
                \caption{单透镜与消色差双胶合的色差差异}
                \label{fig:achromat_edmund_main}
            \end{figure}
        \end{column}
        \hfill
        \begin{column}{.56\textwidth}
            \textbf{消色差的基本逻辑：}
            \begin{itemize}
                \item 正透镜提供主要光焦度，但会产生色焦移。
                \item 负透镜选用不同色散玻璃，用相反色散贡献抵消。
                \item 常见组合是低色散冕牌玻璃与高色散火石玻璃。
            \end{itemize}

            \begin{block}{在 Zemax 中的落点}
                \begin{itemize}
                    \item 多波长权重决定系统偏向哪个谱段。
                    \item 玻璃目录决定可选材料空间。
                    \item Ray Fan 和 OPD 可判断色差是否真正改善。
                    \item MTF/PSF 确认多波长综合像质。
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Glass Substitution Workflow}
\begin{frame}{玻璃替换：直接在真实玻璃目录中优化}
    \begin{columns}[T]
        \begin{column}{.45\textwidth}
            \begin{enumerate}
                \item 在 LDE 中将玻璃 Solve 类型设为 \alert{Substitute}。
                \item 选择供应商目录，如 Schott、Hoya、Ohara、CDGM。
                \item 构建 RMS Wavefront 或 MTF/Spot 相关评价函数。
                \item 使用 Hammer 或 Global Search 处理离散玻璃变化。
                \item 用玻璃替换模板过滤成本、状态和耐候性。
            \end{enumerate}

            \begin{alertblock}{关键}
                玻璃型号是离散变量，Local Optimization 不擅长跨越材料跳变；全局搜索更适合玻璃替换。
            \end{alertblock}

\subsection{公差分析：从名义设计到真实制造}

\englishtitle{Tolerance Analysis}
\begin{frame}{为什么名义设计之后必须做公差分析}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{block}{名义设计的限制}
                名义设计默认曲率、厚度、折射率、偏心、倾斜都精确无误；真实制造中这些参数都会偏离。没有公差分析，就无法判断样品做出来以后还能不能满足指标。
            \end{block}

            \textbf{公差分析回答四个问题：}
            \begin{enumerate}
                \item 哪些面或参数最敏感？
                \item 当前供应商能力是否足够？
                \item 是否需要设置补偿器？
                \item 预计良率与成本是否可接受？
            \end{enumerate}
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \scriptsize
            \begin{table}
