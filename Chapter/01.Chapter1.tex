\englishtitle{One Story for Aberration Analysis}
\begin{frame}{本报告的主线}
\label{1.1}
        \begin{center}
            \fbox{\textbf{理想成像}}\quad
            $\rightarrow$\quad
            \fbox{\textbf{OPD/波前}}\quad
            $\rightarrow$\quad
            \fbox{\textbf{Ray Fan/Zernike}}\quad
            $\rightarrow$\quad
            \fbox{\textbf{PSF/MTF}}
        \end{center}

        \bigbreak
        \begin{itemize}
            \item \alert{理想成像}给出参考：物点应成像为像点。
            \item \alert{OPD}描述实际波前偏离参考球面的程度。
            \item \alert{Ray Fan}从光线落点诊断像差形状。
            \item \alert{Zernike}把波前拆成可量化的正交模式。
            \item \alert{PSF/MTF}回答最终图像是否足够清晰。
        \end{itemize}
        \hfil
        \begin{alertblock}{总结}
            从波前误差出发，用 Ray Fan 与 Zernike 找到像差来源，再用 Zemax 把设计推向可制造的像质目标。
        \end{alertblock}
\end{frame}

\englishtitle{First-Order Layout, Third-Order Aberration and Zernike}
\begin{frame}{一阶、三阶与 Zernike 的关系}
\label{1.2}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \textbf{一阶设计量回答：系统应该把像成在哪里？}
            \begin{itemize}
                \item 有效焦距 $f'$、放大率 $\beta$、像高 $y'$
                \item 孔径、$F/\#$、NA、视场角 $\omega$
                \item 工作距离、后焦距、系统总长
            \end{itemize}

            \begin{equation}
                \beta = \frac{y'}{y}, \qquad
                J = nyu = n'y'u'
                \label{eq:first_order_main}
            \end{equation}

            \begin{block}{拉格朗日不变量J}
                $J$ 是\alert{拉格朗日不变量}（也称光学不变量），表示物方与像方的“孔径 $\times$ 视场”守恒关系。即一阶阶段就已经在约束 NA、视场和倍率之间的可交换空间。
            \end{block}
        \end{column}
        \hfill
        \begin{column}{.52\textwidth}
            \textbf{三阶像差回答：系统偏离理想成像的最低阶形状是什么？}
            \begin{itemize}
                \item 球差：孔径三次项，轴上也存在
                \item 彗差：孔径二次项，随视场一次增长
                \item 像散/场曲：孔径一次项，随视场二次增长
                \item 畸变：主光线像高偏移，主要改变几何相似性
            \end{itemize}

            \begin{block}{Zernike 多项式的角色}
                Seidel 像差给出的是典型像差类别；Zernike 多项式给出的是波前误差的展开坐标。一阶量先决定“像成在哪里、通光量有多大”，三阶像差再说明“偏离理想的最低阶形状”；而要进一步追到\alert{波前本身的误差来源}，就需要 Zernike 这样的正交展开语言。
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\subsection{光程、波前与成像表征}
\label{sec:1.2}

\englishtitle{Optical Path Difference, Wavefront and PSF}
\begin{frame}{光程差、波前误差与点扩散函数}
\label{1.3}
    \begin{columns}[T]
        \begin{column}{.52\textwidth}
            \textbf{光程(Optical Path Length)：}
            \begin{equation}
                OPL = \int n(s)\,ds
                \label{eq:opl_main}
            \end{equation}

            \textbf{光程差(OPD)或波像差：}
            \begin{equation}
                OPD = OPL_{\text{actual}} - OPL_{\text{reference}}
                \equiv W(\rho,\theta)
                \label{eq:opd_main}
            \end{equation}

            \textbf{相位误差：}
            \begin{equation}
                \Delta\phi(\rho,\theta)
                = \frac{2\pi}{\lambda}W(\rho,\theta)
                \label{eq:phase_main}
            \end{equation}

            \begin{equation}
                PSF=\left|\mathcal{F}\{P(\rho,\theta)\}\right|^2,
                \qquad i=o\otimes PSF
                \label{eq:psf_conv_main}
            \end{equation}

            \begin{block}{理想成像的波前条件}
                从同一物点出发并到达同一理想像点的所有光线，应具有相同光程；等价地，实际波前应与以理想像点为球心的参考球面重合。
            \end{block}
        \end{column}
        \hfill
        \begin{column}{.48\textwidth}


            \textbf{广义光瞳函数：}
            \begin{equation}
                P(\rho,\theta)=A(\rho,\theta)
                \exp\!\left[i\frac{2\pi}{\lambda}W(\rho,\theta)\right]
                \label{eq:pupil_main}
            \end{equation}

            \begin{block}{什么是广义光瞳函数}
                $A(\rho,\theta)$ 描述振幅透过率，指数项描述 OPD 引起的相位延迟；因此它把\alert{振幅信息和相位信息}统一到同一个复函数里。
            \end{block}

            \begin{alertblock}{光程差作为统一描述量的意义}
                球差、彗差、像散等几何像差，本质上都对应不同孔径或不同视场光线的\alert{光程不再相等}。因此，OPD 是把传统像差分类与最终像质退化联系起来的统一物理量。
            \end{alertblock}

            % \begin{block}{为什么光学设计里 OPD 很重要}
            %     从 OPD 可以直接走到相位误差、波前图、Zernike、PSF 和 MTF；所以它正处在“结构误差”和“图像后果”之间，是 Zemax 里最核心的中间物理量。
            % \end{block}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{PSF, OTF and Image Formation}
\begin{frame}{NA、FWHM 与 PSF 尺度}
\label{1.4}
    \begin{columns}[T]
        \begin{column}{.46\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=.94\textwidth,height=.55\textheight,keepaspectratio]{Figure/PSF_01.jpg}
                \caption{PSF、FWHM 与 NA 的关系}
                \label{fig:psf_na_main}
            \end{figure}

            \vspace{-4pt}

            \begin{equation}
                r_{\text{Airy}} = \frac{0.61\lambda}{NA}, \qquad
                \mathrm{FWHM}\propto \frac{\lambda}{NA}
                \label{eq:airy_otf_main}
            \end{equation}
        \end{column}
        \hfill
        \begin{column}{.54\textwidth}
            \begin{block}{NA 变大时会发生什么}
                数值孔径 NA 越大，系统能接收的高角度衍射分量越多，因此焦点能量压缩得更紧，Airy 主瓣半径变小，FWHM 也随之减小。
            \end{block}

            \begin{itemize}
                \item \textbf{NA 变大} $\Rightarrow$ 主瓣更窄，点像更集中，分辨能力更强。
                \item \textbf{FWHM 变小} $\Rightarrow$ 半高宽减小，表示系统对点目标的扩散更弱。
                \item \textbf{PSF 变好} $\Rightarrow$ 能量更多集中在中心瓣，而不是扩散到外围。
                \item 当 OPD 增大时，这种集中性被破坏，PSF 会展宽并变得不对称。
            \end{itemize}

            \begin{alertblock}{和像质的关系}
                对成像系统来说，更小的 FWHM 意味着两个相邻点更不容易互相重叠；因此 NA 提升通常直接带来更高分辨率和更清晰的点像。
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{PSF Convolution and Blurred Image Formation}
\begin{frame}{PSF 卷积：图像是怎样被模糊的}
\label{1.4a}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=.98\textwidth,height=.50\textheight,keepaspectratio]{Figure/PSF_02.jpg}
                \caption{物体与 PSF 卷积形成退化图像}
                \label{fig:psf_conv_main}
            \end{figure}

            \vspace{-4pt}
            \begin{equation}
                i(x,y)=o(x,y)\otimes PSF(x,y)
            \end{equation}
        \end{column}
        \hfill
        \begin{column}{.52\textwidth}
            \begin{block}{卷积在物理上是什么意思}
                把物体上的每一个点，都看成一个点光源；每个点都会在像面上生成一个自己的 PSF。最终图像，就是所有这些 PSF 按照原物亮度加权后叠加的结果。
            \end{block}

            \begin{itemize}
                \item 第一步：取物体上的一个亮点，把 PSF 中心放到这个点对应的位置。
                \item 第二步：按这个亮点的强度，对整张 PSF 做幅值缩放。
                \item 第三步：对物体上所有点重复这个过程。
                \item 第四步：把所有平移并缩放后的 PSF 相加，得到最终模糊图像。
            \end{itemize}

            \begin{alertblock}{PSF 展宽会让图像变糊}
                每个点不再只影响一个像素附近，而会把能量散到更大的区域；相邻结构的 PSF 一旦重叠，就会导致边缘变软、细节对比下降
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Confocal Microscopy: Two Convolutions and Resolution Boost}
\begin{frame}{共聚焦：为什么卷积两次，分辨率提升 $\sqrt{2}$}
\label{1.4b}
    \small
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \textbf{宽场成像：一次卷积}
            \begin{equation}
                i_{\text{wide}}(x,y)=o(x,y)\otimes PSF_{\text{sys}}(x,y)
                \label{eq:widefield_main}
            \end{equation}

            \textbf{共聚焦成像：两次卷积}
            \begin{equation}
                i_{\text{conf}}(x,y)=o(x,y)\otimes
                \bigl[PSF_{\text{ill}}(x,y)\;\cdot\;PSF_{\text{det}}(x,y)\bigr]
                \label{eq:confocal_main}
            \end{equation}

            \vspace{-2pt}
            \begin{block}{为什么是两次}
                \begin{enumerate}
                    \item \alert{照明 PSF}：激发光被物镜聚焦到样品上的\alert{一个点}，照明本身就是一个 PSF 分布的衍射斑。
                    \item \alert{探测 PSF}：荧光信号通过共聚焦针孔后再次经物镜成像，针孔只让焦点附近的信号通过。
                    \item 有效 PSF 是二者的乘积——针孔\alert{物理上切掉了离焦和离轴信号}，不是后期算出来的。
                \end{enumerate}
            \end{block}
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \textbf{同一物镜照明+探测：}
            \begin{equation}
                PSF_{\text{eff}} = PSF_{\text{ill}} \times PSF_{\text{det}}
                \approx PSF \times PSF
                \label{eq:psf_product}
            \end{equation}

            \textbf{高斯近似下的直观结论：}
            \begin{equation}
                e^{-\frac{r^2}{2\sigma^2}} \!\cdot\!
                e^{-\frac{r^2}{2\sigma^2}}
                = e^{-\frac{r^2}{2(\sigma/\sqrt{2})^2}}
                \label{eq:gaussian_product}
            \end{equation}

            \vspace{-2pt}
            \begin{alertblock}{$\sqrt{2}$ 倍分辨率提升的本质}
                两个相同 $\sigma$ 的高斯函数相乘，等效宽度变为 $\sigma/\sqrt{2}$。
                反映到成像上，\alert{有效 PSF 的横向半高宽缩小约 $\sqrt{2}$ 倍}，这就是共聚焦比宽场分辨率更好的物理根源。
            \end{alertblock}

            \begin{itemize}
                \item \alert{有效点扩散函数更窄}。
                \item 针孔越大，乘积越退化为单次卷积，$\sqrt{2}$ 优势逐渐消失。
                % \item Airy 斑的精确形式下，乘积仍给出类似的窄化效果。
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\subsection{Seidel 像差与 Zernike 展开}

\englishtitle{Third-Order Aberration as Residual After Paraxial Imaging}
\begin{frame}{三阶像差：理想成像之后的最低阶残差}

    \begin{equation}
        \Delta y(h,\omega)
        \approx A h
        + B h^3
        + C h^2\tan\omega
        + D h\tan^2\omega
        + E\tan^3\omega
        \label{eq:rayfan_third_order}
    \end{equation}

    \begin{table}

\englishtitle{Zernike Polynomial Expansion}
\begin{frame}{Zernike 多项式：把波前拆成标准模式}
\label{1.8}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \scriptsize
            对圆形光瞳，波像差通常展开为：
            \begin{equation}
                W(\rho,\theta)=\sum_j C_j Z_j(\rho,\theta),
                \qquad 0\le \rho \le 1
                \label{eq:zernike_main}
            \end{equation}

            \begin{equation}
                Z_n^m(\rho,\theta)
                =R_n^m(\rho)\cos(m\theta)
                \ \text{或}\
                R_n^m(\rho)\sin(m\theta)
                \label{eq:zernike_nm_main}
            \end{equation}

            \begin{block}{什么是泽尼克多项式}
                Zernike 多项式是在\alert{单位圆光瞳}上定义的一组正交基函数。它把复杂波前 $W$ 拆成若干个标准模式，因此每个系数 $C_j$ 都可解释为某类像差的强弱。
            \end{block}

            \vspace{-2pt}
            \begin{block}{为什么它在光学里特别常用}
                圆孔径天然匹配光瞳；低阶模式与 Seidel 像差有自然对应；正交性使系数分解、拟合和数值计算更稳定。
            \end{block}

            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item $\rho$：归一化孔径半径，中心为 0，边缘为 1。
                \item $\theta$：方位角，决定模式的朝向。
                \item $n$：径向阶数，越大表示径向起伏越复杂。
                \item $m$：角向频率，决定图案对称性与瓣数。
                \item $C_j$：该模式的系数，大小表征贡献，正负表征波前方向。
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{.52\textwidth}
            \begin{center}
                \includegraphics[width=.74\linewidth,height=.35\textheight,keepaspectratio]{Figure/Zernike}
            \end{center}
            \vspace{-4pt}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 红/蓝可理解为相对参考球面的正负波前偏差。
                \item \textbf{Piston} 是整体相位常数，\alert{不影响清晰度}；\textbf{Tilt X/Y} 是线性倾斜，对应像点平移或装调倾斜。
                \item \textbf{Defocus} 表示离焦；两项 \textbf{Astigmatism} 表示两个正交方向焦点不同。
                \item \textbf{Coma} 表示轴外不对称拖尾；两侧三叶形常对应 \textbf{Trefoil}。
                \item \textbf{Primary spherical} 表示中心与边缘光线聚焦不一致；其余图样可视为更高阶复杂模式。
            \end{itemize}

            \vspace{-2pt}
            \begin{alertblock}{常见两套体系}
                常见有 \textbf{Standard Zernike} 和 \textbf{Fringe Zernike}。二者物理模式相近，但\alert{编号、归一化和符号习惯可能不同}，比较系数前必须先确认所用体系。
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Zernike Modes and Physical Aberrations}
\begin{frame}{常用 Zernike 模式、Seidel 对应与编号体系}
\label{1.9}
    \tiny
    \begin{table}
        \centering
        \renewcommand{\arraystretch}{1.22}
        \begin{tabular}{p{2.0cm}|p{2.8cm}|p{2.6cm}|p{3.2cm}}
            \toprule
            \textbf{Zernike 模式} & \textbf{典型形式} & \textbf{物理含义} & \textbf{诊断用途} \\
            \midrule
            Piston & $1$ & 整体相位常数 & 不影响成像，通常去除 \\
            Tilt X/Y & $\rho\cos\theta,\ \rho\sin\theta$ & 像点平移或装调倾斜 & 通常作为对准误差处理 \\
            Defocus & $2\rho^2-1$ & 像面前后移动 & 与 Ray Fan 直线斜率对应 \\
            Astigmatism & $\rho^2\cos2\theta,\ \rho^2\sin2\theta$ & 两个正交方向焦点不同 & 区分子午与弧矢焦面 \\
            Coma & $(3\rho^3-2\rho)\cos\theta$ 等 & 轴外不对称波前 & 对应 Ray Fan 二次弯曲 \\
            Primary spherical & $6\rho^4-6\rho^2+1$ & 边缘与近轴光线焦点不同 & 对应轴上三次型 Ray Fan \\
            Higher order terms & $n\ge5$ & 高阶球差、二级彗差等 & 大 NA、自由曲面或复杂系统中需关注 \\
            \bottomrule
        \end{tabular}
        \caption{低阶 Zernike 模式与常见物理像差之间的联系}
        \label{tab:zernike_modes_main}
    \end{table}

    \vspace{-4pt}
    \begin{block}{与 Seidel 的关系}
        在低阶、圆孔径和近轴条件下，defocus、astigmatism、coma、spherical 与经典 Seidel 类别有自然对应；但 Zernike 是\alert{波前展开基底}，Seidel 是\alert{像差分类}，二者并不是严格一一等价。
    \end{block}

    \vspace{-4pt}
    \begin{alertblock}{Standard 与 Fringe}
        Zemax 中常见的是 \textbf{Zernike Standard Coefficients}；另一套常用体系是 \textbf{Fringe Zernike}。做跨软件、跨文献比较时，要先核对编号顺序和归一化方式。
    \end{alertblock}

    \vspace{-4pt}
    \begin{block}{使用建议}
        先移除 piston、tilt 和必要的 defocus，再讨论剩余 Zernike 系数；否则装调误差和像面位置会掩盖真正的设计像差。
    \end{block}
\end{frame}

            \begin{alertblock}{设计分析要点}
                色差分析应与材料选择同步展开：先用阿贝图确定可行材料空间，再结合 Ray Fan 与 OPD 判断色差残差是否满足成像要求。
