\section{Zemax光学设计实践}

\subsection{设计指标、操作数与流程}

\englishtitle{From Design Specs to an Initial Optical Layout}
\begin{frame}{开始 Zemax 设计前先定义什么？}
    \begin{columns}[T]
        \begin{column}{.52\textwidth}
            \textbf{先写清楚系统规格：}
            \begin{itemize}
                \item 焦距 / 放大率 / 工作距离
                \item 孔径或 $F/\#$、入瞳位置
                \item 视场角、像面尺寸、探测器像元
                \item 设计波长、权重、是否宽光谱
                \item 后工作距(BFL)、总长(TTL)、封装空间
                \item 允许的畸变、照度下降、MTF指标
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{.48\textwidth}
            \begin{block}{在 OpticStudio 中通常对应}
                \begin{itemize}
                    \item \textbf{System Explorer}：波长、视场、孔径、单位
                    \item \textbf{Lens Data Editor}：曲率、厚度、材料、光阑、面型
                    \item \textbf{Merit Function Editor}：把设计目标写成可优化的数学约束
                \end{itemize}
            \end{block}

            \alert{经验：}如果规格书里没有先说明“最重要的是分辨率、照度、体积还是成本”，后续优化很容易陷入来回拉扯。
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Merit Function and Optimization Operands}
\begin{frame}{Zemax 中的操作数是什么？}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{block}{评价函数(Merit Function)}
                Zemax 用一组\alert{操作数(Operands)}表示设计目标。每一行操作数都在问系统一个问题，例如焦距是否正确、像高是否命中、MTF 是否达标、厚度是否可制造。
            \end{block}

            \textbf{常用思路：}
            \begin{itemize}
                \item 先用\alert{一阶性能}把系统“拉到正确位置”
                \item 再用\alert{像差/像质}把成像做清晰
                \item 最后用\alert{边界约束}保证结构可加工、可装调
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \textbf{常见操作数分类：}
            \begin{itemize}
                \item \texttt{EFFL}, \texttt{TOTR}, \texttt{WFNO}/\texttt{TFNO}：焦距、总长、工作光圈
                \item \texttt{REAY}/\texttt{REAX}, \texttt{TRAY}/\texttt{TRAX}：像高与横向像差
                \item \texttt{AXCL}, \texttt{DIMX}：轴向色差与畸变
                \item \texttt{RSCE}, \texttt{RWCE}/\texttt{RWCH}, \texttt{MTFA}, \texttt{DENC}：光斑、波前、MTF、圈入能量
                \item \texttt{MNCT}/\texttt{MNCG}, \texttt{MNET}/\texttt{MNEG}, \texttt{DMVA}/\texttt{DMLT}：厚度、边缘厚度、口径约束
            \end{itemize}

            \alert{操作数不是越多越好，而是要避免重复约束同一物理量。}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{How to Build a Merit Function}
\begin{frame}{操作数怎么组织才高效？}
    \small
    \begin{table}
        \centering
        \setlength{\tabcolsep}{4pt}
        \renewcommand{\arraystretch}{1.35}
        \begin{tabular}{p{2.1cm}|p{3.3cm}|p{5.5cm}}
            \toprule
            \textbf{层级} & \textbf{目标} & \textbf{建议操作数} \\
            \midrule
            一阶规格 & 定住焦距、像高、总长、孔径 & \texttt{EFFL}, \texttt{REAY}, \texttt{TOTR}, \texttt{WFNO}/\texttt{TFNO} \\
            像差控制 & 压低球差、彗差、场曲、色差 & \texttt{TRAY}/\texttt{TRAX}, \texttt{AXCL}, \texttt{DIMX} \\
            像质指标 & 面向分辨率或探测器匹配 & \texttt{RSCE}, \texttt{RWCH}, \texttt{MTFA}, \texttt{DENC} \\
            结构约束 & 保证可制造、可装调 & \texttt{MNCT}, \texttt{MNEG}, \texttt{DMVA}, \texttt{DMLT} \\
            \bottomrule
        \end{tabular}
    \end{table}

    \vspace{4pt}
    \textbf{组织原则：}
    \begin{itemize}
        \item 先低权重放入\alert{一阶目标}，确认系统没有跑偏
        \item 再逐步提高\alert{关键视场、关键波长、关键频率}的权重
        \item 将\alert{制造约束}始终留在评价函数中，避免后期“像质好了但做不出来”
    \end{itemize}
\end{frame}

\englishtitle{Typical Zemax Workflow for Lens Design}
\begin{frame}{Zemax 中的典型设计流程}
    \begin{enumerate}
        \item \textbf{建立规格}：在 \texttt{System Explorer} 中定义波长、视场、孔径、像面和单位。
        \item \textbf{搭建初始结构}：在 \texttt{Lens Data Editor} 中选择初始型，如单透镜、消色差双胶合、对称结构、Cooke triplet 等。
        \item \textbf{设置求解与光阑}：确定 stop 位置，给厚度/焦距/像面添加 solve，使系统能先成像。
        \item \textbf{建立评价函数}：先加一阶规格，再加像差和像质操作数，最后加厚度、边缘厚度、口径等边界约束。
        \item \textbf{执行优化}：先做局部优化(DLS)，复杂系统再考虑全局搜索；每轮优化后检查是否出现“假优解”。
        \item \textbf{分析与收尾}：检查 Ray Fan、Spot、MTF、畸变、色差、照度、容差和热漂移，必要时回到结构层重构。
    \end{enumerate}

    \bigbreak
    \alert{真正高效的流程不是“盲目调参数”，而是“结构选择 $\rightarrow$ 约束设置 $\rightarrow$ 分析闭环”。}
\end{frame}

\englishtitle{Practical Notes for Designing Real Lenses}
\begin{frame}{设计镜头时需要注意什么？}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \textbf{像差与结构层面：}
            \begin{itemize}
                \item 光阑位置会强烈影响\alert{彗差、像散和照度}
                \item 玻璃搭配决定\alert{色差上限}，不要只看折射率，也要看阿贝数和部分色散
                \item 过度依赖单个非球面虽然有效，但会增加制造与检测难度
                \item 大视场系统要同时盯住\alert{场曲、畸变、渐晕和 CRA}
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \textbf{工程实现层面：}
            \begin{itemize}
                \item 始终限制\alert{中心厚度、边缘厚度、最小空气间隔}，避免不可加工结构
                \item 不要只优化中心视场，要覆盖\alert{边缘视场和边缘波长}
                \item 关注像元尺寸、封装公差、温漂和装调偏心，避免“名义设计很好，实物很差”
                \item 最终交付前一定做\alert{容差分析}，否则很难判断量产风险
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
