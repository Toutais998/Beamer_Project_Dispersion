\section{光学像差理论基础}

\subsection{像差概念与波像差理论}

\englishtitle{What is Aberration?}
\begin{frame}{什么是像差？}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \textbf{像差(Aberration)}的本质是\alert{会聚差异}——光线无法会聚于理想像点。

            \bigbreak
            理想的\alert{无像差镜头}应具备：
            \begin{itemize}
                \item 物点 $\rightarrow$ 像点(点 $\rightarrow$ 点)
                \item 直线 $\rightarrow$ 直线
                \item 平面 $\rightarrow$ 平面
                \item 无色散(不受波长影响)
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{.52\textwidth}
            \textbf{像差的分类：}
            \begin{itemize}
                \item \alert{单色像差}：即使入射单色光也会产生
                \begin{itemize}
                    \item 球差、彗差、像散、场曲、畸变
                \end{itemize}
                \item \alert{色差}：因波长依赖性导致
                \begin{itemize}
                    \item 轴向色差(位置色差)
                    \item 倍率色差(横向色差)
                \end{itemize}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Optical Path and Wavefront Deviation}
\begin{frame}{光程与波前偏离的物理定义}
    \begin{columns}[T]
        \begin{column}{.52\textwidth}
            \textbf{光程(Optical Path Length, OPL)：}
            光在折射率 $n$ 连续分布介质中的积分路径：
            \begin{equation}
                OPL = \int n(s)\,ds
                \label{eq:opl}
            \end{equation}

            \textbf{光程差(OPD)：}
            两束光线的 OPL 之差，等效为\alert{波像差函数} $W(x,y)$：
            \begin{equation}
                OPD = OPL_{\text{实际}} - OPL_{\text{理想}} \equiv W(x,y)
                \label{eq:opd}
            \end{equation}
        \end{column}
        \hfill
        \begin{column}{.48\textwidth}
            \begin{block}{马吕斯定律}
                理想成像系统中，物点到像点的所有光线光程均相等。
            \end{block}

            \textbf{波像差的物理意义：}
            \begin{itemize}
                \item 实际波前与理想参考球面波前的\alert{法向几何偏离量}
                \item $W(x,y) \neq 0$ 时，点像退化为弥散斑
                \item 波像差是评估像质的\alert{根本物理量}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{From Phase Modulation to Image Plane}
\begin{frame}{从相位调制到像面光场}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{Figure/图片1.png}
                \caption{有像差与无像差的PSF对比}
                \label{fig:psf_compare}
            \end{figure}
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \textbf{相位差：}
            \begin{equation}
                \Delta\phi = k \cdot OPD = \frac{2\pi}{\lambda} W(x,y)
                \label{eq:phase_diff}
            \end{equation}

            \textbf{广义光瞳函数：}
            \begin{equation}
                P(x,y) = A(x,y) \exp\!\left[i\frac{2\pi}{\lambda}W(x,y)\right]
                \label{eq:pupil_func}
            \end{equation}

            \textbf{点扩散函数(Point Spread Function, PSF)：}
            \begin{equation}
                PSF(u,v) = \left|\iint P(x,y)\,e^{-i\frac{2\pi}{\lambda f}(ux+vy)}dx\,dy\right|^2
                \label{eq:psf}
            \end{equation}

            \alert{核心关系链：} OPD $\rightarrow$ 相位 $\rightarrow$ PSF $\rightarrow$ 像质
        \end{column}
    \end{columns}
\end{frame}

\subsection{点扩散函数与成像模型}

\englishtitle{Point Spread Function and Airy Disk}
\begin{frame}{点扩散函数(PSF)与艾里斑}
    \begin{columns}[T]
        \begin{column}{.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.95\textwidth]{Figure/PSF_01.png}
                \caption{不同NA下的PSF——衍射图案随NA增大而收缩}
                \label{fig:psf_na}
            \end{figure}
        \end{column}
        \hfill
        \begin{column}{.5\textwidth}
            \textbf{PSF 的定义：}
            \bigbreak
            从样本中无限小点光源发射、通过高数值孔径($NA$)物镜传输到像平面的\alert{三维衍射图案}，是图像形成模型的\alert{基本单位}。

            \bigbreak
            \textbf{艾里斑(Airy Disk)：}
            在 $x$-$y$ 平面上形成中心明亮、周围环绕同心光环的衍射图案。中心亮斑半径由 $NA$ 决定：

            \begin{equation}
                r_{\text{Airy}} = \frac{0.61\,\lambda}{NA}
                \label{eq:airy_radius}
            \end{equation}

            \alert{NA 越大，PSF 越集中，分辨率越高。}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Resolution and Rayleigh Criterion}
\begin{frame}{分辨率与瑞利判据}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \textbf{半峰全宽(FWHM)：}
            用来描述 PSF 的宽度，进而定义物镜的\alert{横向分辨率}和瑞利极限。

            \bigbreak
            在没有球差的完美透镜中，点源衍射图案呈\alert{沙漏形}(横向)或\alert{橄榄球形}(轴向)对称分布。

            \bigbreak
            两个相邻发光点能被分辨的最小距离，决定了显微镜的\alert{横向和轴向分辨率}。
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \begin{block}{瑞利判据(Rayleigh Criterion)}
                当第一个艾里斑的中心正好落在第二个艾里斑的第一暗环上时，两点恰好可分辨：
                \begin{equation}
                    d_{\text{min}} = \frac{0.61\,\lambda}{NA}
                    \label{eq:rayleigh}
                \end{equation}
            \end{block}

            \textbf{提高分辨率的途径：}
            \begin{itemize}
                \item 增大 $NA$(高数值孔径物镜)
                \item 减小 $\lambda$(短波长/紫外光源)
                \item 使用反卷积等计算恢复技术
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Convolution Model of Imaging}
\begin{frame}{成像过程的卷积模型}
    \begin{columns}[T]
        \begin{column}{.55\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{Figure/PSF_02.png}
                \caption{图像退化与反卷积恢复过程}
                \label{fig:psf_convolution}
            \end{figure}
        \end{column}
        \hfill
        \begin{column}{.45\textwidth}
            在满足\alert{线性}和\alert{空间平移不变性}的前提下，最终图像 $i(r)$ 是真实物体 $o(r)$ 与系统 $PSF(r)$ 的\alert{卷积}：

            \begin{equation}
                i(x,y,z) = o(x,y,z) \otimes PSF(x,y,z)
                \label{eq:convolution}
            \end{equation}

            \bigbreak
            连续域卷积积分：
            \begin{equation}
                i(x) = \int_{-\infty}^{\infty} o(x-x') \times PSF(x')\,dx'
                \label{eq:conv_integral}
            \end{equation}

            \alert{卷积的含义：}每个物点都被替换为该系统对应的 PSF 图案，所有 PSF 按强度加权叠加形成最终图像。
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Frequency Domain Analysis and OTF}
\begin{frame}{频域分析与光学传递函数(OTF)}
    \begin{columns}[T]
        \begin{column}{.55\textwidth}
            通过\alert{傅里叶变换}将空间坐标转化为空间频率，频域中卷积简化为\alert{逐点乘法}：

            \begin{equation}
                F\{i\} = F\{o\} \times F\{PSF\}
                \label{eq:freq_conv}
            \end{equation}

            \bigbreak
            \textbf{光学传递函数(OTF)：}
            \begin{equation}
                OTF = F\{PSF\}
                \label{eq:otf}
            \end{equation}
            OTF 描述显微系统如何传递不同空间频率的信号。
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \begin{block}{显微镜的低通滤波特性}
                \begin{itemize}
                    \item \textbf{低频}：大尺寸特征，容易通过
                    \item \textbf{中高频}：细节逐渐被衰减
                    \item \textbf{截止频率} $2NA/\lambda$ 以上：极高频完全丢失
                \end{itemize}
            \end{block}

            \alert{衍射极限的本质：}OTF 在高频处截止，超过截止频率的信息不可能被任何光学系统传递。
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Deconvolution Image Restoration}
\begin{frame}{反卷积图像恢复技术}
    \begin{columns}[T]
        \begin{column}{.55\textwidth}
            \textbf{反卷积的数学原理：}
            理论上通过频域逆运算可直接恢复物体信息：
            \begin{equation}
                F\{o\} = \frac{F\{i\}}{OTF}
                \label{eq:deconv}
            \end{equation}

            然而，直接反滤波会\alert{极大放大高频噪声}，使结果完全失真。
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \textbf{主要反卷积算法：}
            \begin{itemize}
                \item \alert{去模糊算法}：如最近邻法——快速但可能降低 SNR
                \item \alert{约束迭代法}：如 Jansson-van Cittert——设定非负约束，有效降噪
                \item \alert{盲反卷积}：无需预先测量 PSF，同时估计物体分布和系统 PSF
            \end{itemize}
        \end{column}
    \end{columns}

    \bigbreak
    \alert{核心权衡：}反卷积需要在\alert{分辨率提升}、\alert{处理速度}和\alert{噪声控制}之间取得平衡。
\end{frame}

\englishtitle{Ray Description in Geometrical Optics}
\begin{frame}{几何光学中的光束描述}
    \begin{columns}[T]
        \begin{column}{.50\textwidth}
            \textbf{光线扇图(Ray Fan)：}
            从物点发出多条光线，形成扇状分布。通过观察光线扇的分布，可以大致了解光线的会聚状态。

            \bigbreak
            \textbf{子午面(Meridional Plane)：}
            包含物点和光轴的平面。光路图通常展示子午面内的光线路径。
        \end{column}
        \hfill
        \begin{column}{.50\textwidth}
            \textbf{弧矢面(Sagittal Plane)：}
            垂直于子午面且包含主光线的平面。

            \bigbreak
            \textbf{主光线：}从任意物点发出的所有光线中，唯一穿过入瞳或孔径光阑中心的光线。
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{First-Order Quantities in Imaging Optics}
\begin{frame}{成像系统的一阶设计量}
    \begin{columns}[T]
        \begin{column}{.56\textwidth}
            在进入像差优化之前，通常要先确定\alert{一阶规格}：

            \begin{itemize}
                \item \textbf{有效焦距 $f'$}：决定成像比例和视场覆盖能力
                \item \textbf{$F/\#$}：近似表征通光能力与景深，$F/\# \approx f'/D_{EP}$
                \item \textbf{视场(Field of View)}：决定离轴像差压力和像面尺寸
                \item \textbf{放大率 $\beta$}：决定物像尺度对应关系
            \end{itemize}

            \begin{equation}
                \beta = \frac{y'}{y}
                \label{eq:magnification}
            \end{equation}

            \alert{设计逻辑：}先满足焦距、孔径、视场与像高，再讨论像差如何压低。
        \end{column}
        \hfill
        \begin{column}{.44\textwidth}
            \begin{block}{Lagrange 不变量}
                \begin{equation}
                    J = nyu = n'y'u'
                    \label{eq:lagrange}
                \end{equation}
            \end{block}

            \textbf{它告诉我们：}
            \begin{itemize}
                \item 视场、孔径和像高之间存在\alert{守恒约束}
                \item 想同时增大视场和数值孔径，系统体积与像差校正难度都会上升
                \item Zemax 中很多优化冲突，本质上都能追溯到这一约束
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\subsection{横向像差图与三阶像差理论}

\englishtitle{What is a Transverse Aberration Diagram?}
\begin{frame}{什么是横向像差图？}
    \begin{columns}[T]
        \begin{column}{.45\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.95\textwidth]{Figure/图片2.png}
                \caption{横向像差图的基本结构}
                \label{fig:trans_aberr_def}
            \end{figure}
        \end{column}
        \hfill
        \begin{column}{.55\textwidth}
            \begin{block}{定义}
                横向像差图是将到达评估面的光线坐标与入射光瞳面上的光线坐标相关联的图表。
            \end{block}

            \begin{itemize}
                \item \textbf{横轴}：光瞳坐标，满刻度对应入瞳直径
                \item \textbf{纵轴}：横向像差量(垂直于光轴方向的偏移)
                \item \textbf{原点}：主光线在评估面上的到达位置
            \end{itemize}

            在几何光学中，沿光轴方向称为\alert{纵向}，垂直于光轴的方向称为\alert{横向}，由此定义的像差称为\alert{横向像差}。
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Geometric Origin of the Ray Fan Plot}
\begin{frame}{横向像差图的几何来源}
    \begin{columns}[T]
        \begin{column}{.62\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{Figure/图片26.png}
                \caption{由入瞳坐标到像面光线穿透点的映射关系}
                \label{fig:rayfan_origin}
            \end{figure}
        \end{column}
        \hfill
        \begin{column}{.38\textwidth}
            \textbf{这张图说明了横向像差图是如何生成的：}
            \begin{itemize}
                \item \textbf{横轴}对应入瞳上的归一化坐标 $h\in[-1,1]$
                \item \textbf{纵轴}对应各条光线在像面的穿透坐标，相对主光线读出
                \item 子午光束形成 \alert{$y$-fan}，弧矢光束可形成 \alert{$x$-fan}
                \item 图上的每个点都是一条真实追迹光线，连线后得到横向像差曲线
            \end{itemize}

            \alert{因此：}横向像差图不是抽象函数图，而是“\alert{入瞳采样} $\rightarrow$ \alert{像面落点}”的直接可视化。
        \end{column}
    \end{columns}
\end{frame}

\englishtitle{Third-Order (Seidel) Aberration Theory}
\begin{frame}{三阶像差理论}
    根据三阶像差理论，横向像差量 $\Delta Y$ 的孔径依赖性和视场依赖性为：

    \begin{equation}
        \Delta Y \propto \mathrm{I}h^3(\tan\omega)^0 - 3\mathrm{II}h^2(\tan\omega)^1 + (3\mathrm{III}+P)h^1(\tan\omega)^2 - \mathrm{V}h^0(\tan\omega)^3
        \label{eq:seidel}
    \end{equation}

    其中 $h$ 为入射光瞳坐标，$\omega$ 为视场角。

    \begin{block}{各系数的物理意义}
        \begin{itemize}
            \item \textbf{I(球差系数)}：$\propto h^3$，与视场角无关
            \item \textbf{II(彗差系数)}：$\propto h^2$，$\propto \tan\omega$
            \item \textbf{III, P(像散/场曲系数)}：$\propto h$，$\propto \tan^2\omega$
            \item \textbf{V(畸变系数)}：与孔径无关，$\propto \tan^3\omega$
        \end{itemize}
    \end{block}
\end{frame}

\englishtitle{Zernike Polynomial Expansion of Wave Aberration}
\begin{frame}{波像差的泽尼克多项式展开}
    \begin{columns}[T]
        \begin{column}{.52\textwidth}
            现代光学设计中，波像差用\alert{泽尼克多项式}在极坐标 $(\rho,\theta)$ 下正交展开：
            \begin{equation}
                W(\rho,\theta) = \sum_{j} C_j Z_j(\rho,\theta)
                \label{eq:zernike}
            \end{equation}
        \end{column}
        \hfill
        \begin{column}{.48\textwidth}
            \begin{itemize}
                \item \textbf{球差：} $W \propto \rho^4$(径向偶次函数)
                \item \textbf{彗差：} $W \propto \rho^3\cos\theta$
                \item \textbf{像散：} $W \propto \rho^2\cos^2\theta$
                \item \textbf{场曲：} $W$ 随视场和孔径平方变化
            \end{itemize}
        \end{column}
    \end{columns}

    \bigbreak
    泽尼克系数 $C_j$ 可以\alert{直接量化}各像差分量强度，方便与 Seidel 像差系数对应转换。
